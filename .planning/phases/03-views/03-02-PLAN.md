    ```tsx
    "use client";

    import { useEffect, useRef, useState } from "react";

    interface CopyButtonProps {
      /** The string written to clipboard on click. */
      value: string;
      /** Visible label in idle state. Default: "⧉ copy". */
      idleLabel?: string;
      /** Visible label after a successful copy. Default: "copied ✓". */
      copiedLabel?: string;
      /** REQUIRED — plain-noun label for screen readers (e.g. "Copy stack JSON"). */
      ariaLabel: string;
      /** Optional className for view-specific composition (.stack-copy, .contact-copy, .copy-button--icon, etc.). */
      className?: string;
    }

    /**
     * Phase 3's ONLY client island (per Pitfall 9 / SHELL-02).
     * Used by stack-view (copy JSON), contact-view (copy email), shipped-view (copy store URL).
     */
    export function CopyButton({
      value,
      idleLabel = "⧉ copy",
      copiedLabel = "copied ✓",
      ariaLabel,
      className
    }: CopyButtonProps) {
      const [copied, setCopied] = useState(false);
      const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);

      useEffect(() => {
        return () => {
          if (timerRef.current !== null) {
            clearTimeout(timerRef.current);
          }
        };
      }, []);

      async function handleClick() {
        try {
          await navigator.clipboard.writeText(value);
        } catch {
          // Clipboard write can fail (permissions, insecure context). Swallow silently —
          // the user-visible UI does not flip to "copied" so failure is implicit.
          return;
        }
        setCopied(true);
        if (timerRef.current !== null) {
          clearTimeout(timerRef.current);
        }
        timerRef.current = setTimeout(() => {
          setCopied(false);
          timerRef.current = null;
        }, 1500);
      }

      const cls = ["copy-button", copied ? "copy-button--confirmed" : null, className]
        .filter(Boolean)
        .join(" ");

      return (
        <button
          type="button"
          className={cls}
          onClick={handleClick}
          aria-label={ariaLabel}
        >
          <span aria-hidden="true">{copied ? copiedLabel : idleLabel}</span>
          <span role="status" aria-live="polite" className="sr-only">
            {copied ? "Copied to clipboard" : ""}
          </span>
        </button>
      );
    }
    ```

    Per D-05/D-06/D-07: `"use client"` first line, default labels `⧉ copy` / `copied ✓`, 1500ms duration, polite aria-live region inside the button using the existing `.sr-only` class (already in globals.css line 202). Confirmed-state class `copy-button--confirmed` is toggled on the button so Plan 03-04 can paint accent text via CSS.

    DO NOT introduce a toast component. DO NOT animate the label swap. DO NOT add a copy-failed UI state in v1 (the catch is intentional silence).

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| view code → user clipboard | clipboard write is a permissioned browser API; user-controlled |
| client island ↔ DOM | useState + setTimeout side effects; React renders the swap |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-04 | Information Disclosure (clipboard write of unintended value) | `CopyButton.value` prop | mitigate | `value` is caller-supplied from typed Profile/Stack/ShippedApp data — never from URL/user input. View-level callers in Wave 4 pass static strings (`PROFILE.email`, `JSON.stringify(STACK,...)`, `app.appStoreUrl`). |
| T-03-05 | Tampering (XSS via dynamic label) | `idleLabel` / `copiedLabel` / `ariaLabel` props | accept | All three props render as text content (React auto-escapes). No `dangerouslySetInnerHTML`. Caller controls labels — no untrusted source. |
| T-03-06 | Denial of Service (rapid-click memory leak from setTimeout) | `setCopied` timeout chain | mitigate | `useRef`-tracked timer is cleared on every click and on unmount via `useEffect` cleanup. Bounded resource use. |
| T-03-07 | Information Disclosure (clipboard.writeText failure leaks via console) | `try/catch` around the async write | mitigate | The catch swallows silently — no `console.error` (matches CONVENTIONS.md zero-console-calls rule). UI signals failure implicitly by not flipping to "copied". |
</threat_model>
