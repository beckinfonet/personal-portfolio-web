```ts
import type { NextConfig } from "next";

const securityHeaders = [
  { key: "Strict-Transport-Security", value: "max-age=63072000; includeSubDomains; preload" },
  { key: "X-Content-Type-Options", value: "nosniff" },
  { key: "Referrer-Policy", value: "strict-origin-when-cross-origin" },
  { key: "X-Frame-Options", value: "DENY" },
  { key: "Permissions-Policy", value: "camera=(), microphone=(), geolocation=()" }
];

const engineerHeaders = [
  // x-portfolio-source deferred to Phase 7 per D-13 (revised) — set when public deploy URL finalized.
  { key: "x-built-with", value: "nextjs-15-react-19" }
];

const nextConfig: NextConfig = {
  async headers() {
    return [
      {
        source: "/:path*",
        headers: [...securityHeaders, ...engineerHeaders]
      }
    ];
  }
};

export default nextConfig;
```

Conventions applied:
- `import type { NextConfig } from "next"` first (preserved)
- Module-level `const securityHeaders` and `const engineerHeaders` arrays before the config (new pattern, idiomatic per Next.js docs cited in RESEARCH.md §2)
- 2-space indent, double quotes, semicolons, no trailing commas (matches CONVENTIONS.md)
- Default export preserved (framework convention)
- The `source: "/:path*"` matcher applies headers to every route including `/`, `/api/*`, static assets, error pages

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser ↔ Next.js runtime | Every HTTP response crosses this boundary; security headers tell the browser how to treat the response (clickjacking, MIME sniffing, sensor APIs, referrer leakage, transport security). |
| Public CDN ↔ user agent | HSTS preload list participation (`includeSubDomains; preload`) commits the domain to HTTPS-only on browser-shipped lists — defense in depth against downgrade attacks. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-INFRA-04a | T (tampering — clickjacking) | Site can be embedded in iframe by malicious page; click-jacks user actions | mitigate | `X-Frame-Options: DENY` — refuses all frame embedding |
| T-INFRA-04b | T (tampering — MIME sniffing) | Browser ignores Content-Type and sniffs response body, executing as script | mitigate | `X-Content-Type-Options: nosniff` |
| T-INFRA-04c | I (info disclosure — referrer leakage) | Full URL with query string sent in `Referer` to external sites (e.g. analytics, link clicks) | mitigate | `Referrer-Policy: strict-origin-when-cross-origin` — same-origin gets full URL, cross-origin gets only origin |
| T-INFRA-04d | T (tampering — sensor API abuse) | Malicious script accesses camera/microphone/geolocation without explicit allow | mitigate | `Permissions-Policy: camera=(), microphone=(), geolocation=()` — empty allowlist denies all origins |
| T-INFRA-04e | T (tampering — HTTP downgrade attack) | Active MITM downgrades connection to plain HTTP, intercepts traffic | mitigate | `Strict-Transport-Security: max-age=63072000; includeSubDomains; preload` — 2-year HSTS with subdomain coverage and preload list eligibility |
| T-INFRA-04f | T (tampering — XSS via inline scripts) | Inline `<script>` tags or `eval` bypass any future CSP | accept (Phase 1) | CSP DEFERRED per D-15 — requires nonce-based strategy threaded through Phase 2 accent boot script. Captured for v2 hardening milestone. |
</threat_model>
