    ```ts
    // tests/contrast.spec.ts
    // Phase 5 A11Y-07: 4 hues × 2 themes × 7 routes = 56 cells.
    // Each cell seeds localStorage BEFORE first paint so AccentBootstrapScript +
    // next-themes pick up the values synchronously. AxeBuilder filters to color-contrast
    // violations and asserts the array is empty.
    //
    // Failure threshold: any serious/critical color-contrast violation fails the build.
    // Predicted failure (Pitfall 5): amber-on-light at oklch(0.5 0.16 60).
    // Remediation: per-hue chroma overrides ONLY for cells that actually fail (D-22).

    import { test, expect } from "@playwright/test";
    import AxeBuilder from "@axe-core/playwright";
    import { ROUTES } from "@/lib/routes";

    const HUES = [145, 75, 200, 340] as const;       // matrix, amber, cyan, magenta
    const THEMES = ["dark", "light"] as const;

    for (const theme of THEMES) {
      for (const hue of HUES) {
        for (const route of ROUTES) {
          test(`contrast: theme=${theme} accent=${hue} route=${route.pathname}`, async ({ page }) => {
            // Seed localStorage BEFORE first paint so:
            //   - AccentBootstrapScript reads "portfolio-accent" and sets --accent-hue
            //   - next-themes reads "theme" and sets <html data-theme="dark|light">
            // Both are pre-paint so the rendered page has the right tokens by the time
            // axe runs.
            await page.addInitScript(
              ([t, h]) => {
                try {
                  window.localStorage.setItem("theme", t);
                  window.localStorage.setItem("portfolio-accent", String(h));
                } catch {
                  /* localStorage unavailable — skip; CSS defaults will apply */
                }
              },
              [theme, hue] as const
            );

            await page.goto(route.pathname);

            // SANITY (Pitfall 10): verify the seed actually applied. Without this,
            // a silent localStorage failure or a future AccentBootstrapScript regression
            // would produce 56 identical axe results. The assertion catches the failure
            // mode before axe runs.
            await expect(page.locator("html")).toHaveAttribute("data-theme", theme);

            const results = await new AxeBuilder({ page })
              .withTags(["wcag2aa", "wcag21aa"])
              .analyze();

            const contrastViolations = results.violations.filter(
              (v) => v.id === "color-contrast"
            );

            // Print full violation details on failure for fast remediation.
            if (contrastViolations.length > 0) {
              console.error(
                `Contrast failures @ ${theme}/${hue}/${route.pathname}:`,
                contrastViolations.map((v) => ({
                  id: v.id,
                  impact: v.impact,
                  nodes: v.nodes.map((n) => ({
                    target: n.target,
                    summary: n.failureSummary
                  }))
                }))
              );
            }

            expect(contrastViolations).toEqual([]);
          });
        }
      }
    }
    ```

    Key constraints:
    - HUES order: matrix(145), amber(75), cyan(200), magenta(340) — UI-SPEC §Color §"Contrast contract" matrix
    - THEMES: only "dark" and "light" (default theme is dark per Phase 2 D-07; light is the second supported theme)
    - `page.addInitScript` runs BEFORE the page's own scripts on every load — the only safe place to seed localStorage so AccentBootstrapScript's pre-paint read sees the right values (Pitfall 10)
    - `expect(page.locator("html")).toHaveAttribute("data-theme", theme)` is non-negotiable — it's the canary that protects against silent localStorage seed failures (Pitfall 10 explicitly calls this out)
    - `withTags(["wcag2aa", "wcag21aa"])` — both for completeness; UI-SPEC §"Contrast contract" cites WCAG 2.1 AA explicitly
    - Filter to `id === "color-contrast"` — Phase 5 only owns contrast (A11Y-07); other axe rules (alt text, etc) are out of scope and may have pre-existing violations not introduced by Phase 5
    - DO NOT use `withRules` or `disableRules` — `withTags` is the simpler API and matches RESEARCH Pattern 7

    Run `npm run test:contrast` against the dev server. Capture the pass/fail per cell — expect 56 tests reported.

    DO NOT edit `playwright.config.ts` here (Plan 05-01 already settled it). DO NOT add new test files (only this one). DO NOT modify `app/globals.css` here — that's Task 2 / Task 3 if remediation is needed.
